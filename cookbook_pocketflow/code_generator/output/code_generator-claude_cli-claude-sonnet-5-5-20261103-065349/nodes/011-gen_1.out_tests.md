def test_basic():
    nums = [2, 7, 11, 15]
    result = two_sum(nums, 9)
    assert sorted(result) == [0, 1]

def test_pair_in_middle():
    nums = [3, 2, 4]
    result = two_sum(nums, 6)
    assert sorted(result) == [1, 2]

def test_duplicates():
    nums = [3, 3]
    result = two_sum(nums, 6)
    assert sorted(result) == [0, 1]

def test_negative_numbers():
    nums = [-1, -2, -3, -4, -5]
    result = two_sum(nums, -8)
    assert sorted(result) == [2, 4]

def test_mixed_signs():
    nums = [-3, 4, 3, 90]
    result = two_sum(nums, 0)
    assert sorted(result) == [0, 2]

def test_zeros():
    nums = [0, 4, 3, 0]
    result = two_sum(nums, 0)
    assert sorted(result) == [0, 3]

def test_target_zero_with_zero_and_other():
    nums = [0, 1, 2]
    result = two_sum(nums, 2)
    assert sorted(result) == [0, 2]

def test_same_element_not_used_twice():
    nums = [5, 1, 9]
    result = two_sum(nums, 10)
    # 5 + 5 would need the same index twice; must not return [0, 0]
    assert result is None or len(result) == 0 or (len(set(result)) == 2 and nums[result[0]] + nums[result[1]] == 10)
    assert not (result and len(result) == 2 and result[0] == result[1])

def test_same_value_twice_valid():
    nums = [5, 1, 5]
    result = two_sum(nums, 10)
    assert sorted(result) == [0, 2]

def test_two_elements():
    nums = [1, 2]
    result = two_sum(nums, 3)
    assert sorted(result) == [0, 1]

def test_pair_at_ends():
    nums = [1, 5, 6, 7, 9]
    result = two_sum(nums, 10)
    assert sorted(result) == [0, 4]

def test_result_sums_to_target():
    nums = [10, 20, 30, 40, 50, 60]
    target = 90
    result = two_sum(nums, target)
    assert len(result) == 2
    assert result[0] != result[1]
    assert nums[result[0]] + nums[result[1]] == target

def test_large_numbers():
    nums = [1000000000, 2000000000, 3000000000]
    result = two_sum(nums, 5000000000)
    assert sorted(result) == [1, 2]

def test_floats():
    nums = [0.5, 1.5, 2.0]
    result = two_sum(nums, 2.5)
    assert sorted(result) == [0, 2]

def test_unsorted_input():
    nums = [8, 1, 5, 3, 7]
    result = two_sum(nums, 12)
    assert sorted(result) == [2, 4]

def test_no_solution():
    nums = [1, 2, 3]
    result = two_sum(nums, 100)
    assert result is None or len(result) == 0

def test_empty_list():
    result = two_sum([], 5)
    assert result is None or len(result) == 0

def test_single_element():
    result = two_sum([5], 5)
    assert result is None or len(result) == 0

def test_input_not_mutated():
    nums = [4, 3, 2, 1]
    copy = list(nums)
    two_sum(nums, 5)
    assert nums == copy

def test_large_list():
    nums = list(range(10000))
    result = two_sum(nums, 9999 + 9998)
    assert sorted(result) == [9998, 9999]
