def test_basic():
    result = two_sum([2, 7, 11, 15], 9)
    assert result is not None
    assert sorted(result) == [0, 1]

def test_basic_middle():
    result = two_sum([1, 5, 3, 7], 8)
    assert result is not None
    assert sorted(result) == [1, 2]

def test_pair_at_end():
    result = two_sum([1, 2, 3, 4, 5], 9)
    assert result is not None
    assert sorted(result) == [3, 4]

def test_two_elements():
    result = two_sum([3, 4], 7)
    assert result is not None
    assert sorted(result) == [0, 1]

def test_duplicates_same_value():
    result = two_sum([3, 3], 6)
    assert result is not None
    assert sorted(result) == [0, 1]

def test_duplicates_in_middle():
    result = two_sum([1, 4, 4, 9], 8)
    assert result is not None
    assert sorted(result) == [1, 2]

def test_zeros():
    result = two_sum([0, 4, 3, 0], 0)
    assert result is not None
    assert sorted(result) == [0, 3]

def test_index_zero_used():
    result = two_sum([5, 1, 2], 7)
    assert result is not None
    assert sorted(result) == [0, 2]

def test_negative_numbers():
    result = two_sum([-1, -2, -3, -4, -5], -8)
    assert result is not None
    assert sorted(result) == [2, 4]

def test_negative_and_positive():
    result = two_sum([-3, 4, 3, 90], 0)
    assert result is not None
    assert sorted(result) == [0, 2]

def test_negative_target_mixed():
    result = two_sum([10, -15, 7, 3], -5)
    assert result is not None
    assert sorted(result) == [0, 1]

def test_large_numbers():
    result = two_sum([1000000000, 2000000000, 3000000000], 5000000000)
    assert result is not None
    assert sorted(result) == [1, 2]

def test_no_solution():
    assert two_sum([1, 2, 3], 10) is None

def test_no_solution_single_use():
    # 2 + 2 = 4 would need the same element twice
    assert two_sum([2, 5, 8], 4) is None

def test_empty_list():
    assert two_sum([], 5) is None

def test_single_element():
    assert two_sum([5], 5) is None

def test_single_element_zero_target():
    assert two_sum([0], 0) is None

def test_does_not_reuse_element():
    result = two_sum([4, 6], 8)
    assert result is None

def test_result_indices_are_distinct():
    result = two_sum([1, 2, 3, 4], 6)
    assert result is not None
    assert result[0] != result[1]

def test_result_sums_to_target():
    nums = [8, 3, 12, 5, 9]
    target = 17
    result = two_sum(nums, target)
    assert result is not None
    assert nums[result[0]] + nums[result[1]] == target

def test_result_indices_in_range():
    nums = [10, 20, 30, 40]
    result = two_sum(nums, 70)
    assert result is not None
    assert 0 <= result[0] < len(nums)
    assert 0 <= result[1] < len(nums)
    assert sorted(result) == [2, 3]

def test_result_has_two_elements():
    result = two_sum([2, 7, 11, 15], 26)
    assert result is not None
    assert len(result) == 2
    assert sorted(result) == [2, 3]
