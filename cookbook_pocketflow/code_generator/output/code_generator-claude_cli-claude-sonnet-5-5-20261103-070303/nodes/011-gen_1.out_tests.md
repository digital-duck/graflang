def test_basic():
    result = two_sum([2, 7, 11, 15], 9)
    assert sorted(result) == [0, 1]

def test_not_first_pair():
    result = two_sum([1, 3, 4, 2], 6)
    assert sorted(result) == [2, 3]

def test_middle_elements():
    result = two_sum([1, 5, 3, 7], 8)
    assert sorted(result) == [1, 2] or sorted(result) == [0, 3]
    assert [1, 5, 3, 7][result[0]] + [1, 5, 3, 7][result[1]] == 8

def test_two_elements():
    result = two_sum([3, 4], 7)
    assert sorted(result) == [0, 1]

def test_duplicates():
    result = two_sum([3, 3], 6)
    assert sorted(result) == [0, 1]

def test_duplicates_in_larger_list():
    nums = [1, 4, 4, 9]
    result = two_sum(nums, 8)
    assert sorted(result) == [1, 2]

def test_negative_numbers():
    nums = [-1, -2, -3, -4, -5]
    result = two_sum(nums, -8)
    assert sorted(result) == [2, 4]

def test_negative_and_positive():
    nums = [-3, 4, 3, 90]
    result = two_sum(nums, 0)
    assert sorted(result) == [0, 2]

def test_zero_target_with_zeros():
    result = two_sum([0, 4, 3, 0], 0)
    assert sorted(result) == [0, 3]

def test_target_zero():
    nums = [5, -5, 1]
    result = two_sum(nums, 0)
    assert sorted(result) == [0, 1]

def test_same_element_not_used_twice():
    nums = [3, 2, 4]
    result = two_sum(nums, 6)
    assert sorted(result) == [1, 2]
    assert result[0] != result[1]

def test_single_element_no_self_pair():
    result = two_sum([5], 10)
    assert result is None or result == [] or result == ()

def test_empty_list():
    result = two_sum([], 5)
    assert result is None or result == [] or result == ()

def test_no_solution():
    result = two_sum([1, 2, 3], 100)
    assert result is None or result == [] or result == ()

def test_large_numbers():
    nums = [10**9, 2, 10**9 - 2, 7]
    result = two_sum(nums, 2 * 10**9 - 2)
    assert sorted(result) == [0, 2]

def test_result_indices_valid():
    nums = [8, 1, 5, 12, 7]
    target = 19
    result = two_sum(nums, target)
    assert len(result) == 2
    assert result[0] != result[1]
    assert 0 <= result[0] < len(nums)
    assert 0 <= result[1] < len(nums)
    assert nums[result[0]] + nums[result[1]] == target

def test_unsorted_input():
    nums = [15, 11, 7, 2]
    result = two_sum(nums, 9)
    assert sorted(result) == [2, 3]

def test_does_not_mutate_input():
    nums = [2, 7, 11, 15]
    copy = list(nums)
    two_sum(nums, 9)
    assert nums == copy

def test_long_list_pair_at_end():
    nums = list(range(1, 1001))
    result = two_sum(nums, 1999)
    assert sorted(result) == [998, 999]

def test_pair_at_start_and_end():
    nums = [1, 2, 3, 4, 5, 9]
    result = two_sum(nums, 10)
    assert sorted(result) == [0, 5]

def test_returns_two_indices():
    result = two_sum([4, 6, 1], 10)
    assert len(result) == 2
